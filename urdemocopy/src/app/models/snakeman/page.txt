// Snakeman Model Component
function SnakemanModel({ animationName }) {
  const modelRef = useRef();
  const { scene, animations } = useGLTF(
    "/models/onepiece/snakeman_animate.glb",
  );
  const { actions } = useAnimations(animations, modelRef);

  useEffect(() => {
    const action = actions[animationName];
    if (action) {
      action.reset().fadeIn(0.3).play();
      // Animation type combo loop settings: use explicit repetition count for single-play
      action.setLoop(
        animationName.includes("combo") ? THREE.LoopOnce : THREE.LoopRepeat,
        animationName.includes("combo") ? 1 : Infinity,
      );
    }
    return () => action?.fadeOut(0.3);
  }, [animationName, actions]);

  return (
    <primitive
      ref={modelRef}
      object={scene}
      scale={185}
      position={[0, -4, 0]}
    />
  );
}

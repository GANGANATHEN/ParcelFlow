"use client";
import React, { useEffect, useRef } from "react";
import gsap from "gsap";

const ParticleMorph = () => {
  const canvasRef = useRef(null);
  const particles = useRef([]);
  const numParticles = 1000; // Kutty kutty shapes evvalavu venum?

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext("2d");
    canvas.width = 600;
    canvas.height = 600;

    // 1. Shapes-oda Points-ah generate panra function
    const getShapePoints = (shapeType) => {
      const points = [];
      if (shapeType === "circle") {
        for (let i = 0; i < numParticles; i++) {
          const angle = Math.random() * Math.PI * 2;
          const radius = Math.random() * 150;
          points.push({
            x: 300 + Math.cos(angle) * radius,
            y: 300 + Math.sin(angle) * radius,
          });
        }
      } else {
        // Square shape points
        for (let i = 0; i < numParticles; i++) {
          points.push({
            x: 150 + Math.random() * 300,
            y: 150 + Math.random() * 300,
          });
        }
      }
      return points;
    };

    const circlePoints = getShapePoints("circle");
    const squarePoints = getShapePoints("square");

    // 2. Initial Particles setup
    particles.current = circlePoints.map((p) => ({
      x: p.x,
      y: p.y,
      originX: p.x,
      originY: p.y,
    }));

    // 3. Animation Loop
    const render = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      ctx.fillStyle = "#3b82f6"; // Tailwind blue-500

      particles.current.forEach((p) => {
        ctx.beginPath();
        // Inga thaan unga "Kutty SVG" logic varum.
        // Ippo namba simple circles draw panrom.
        ctx.arc(p.x, p.y, 2, 0, Math.PI * 2);
        ctx.fill();
      });
      requestAnimationFrame(render);
    };
    render();

    // 4. GSAP Morphing Function
    const morphTo = (targetPoints) => {
      particles.current.forEach((p, i) => {
        gsap.to(p, {
          x: targetPoints[i].x,
          y: targetPoints[i].y,
          duration: 2,
          ease: "expo.inOut",
          delay: Math.random() * 0.5, // Random movement effect-kaga
        });
      });
    };

    // Intervals to switch shapes
    let currentShape = "circle";
    const interval = setInterval(() => {
      if (currentShape === "circle") {
        morphTo(squarePoints);
        currentShape = "square";
      } else {
        morphTo(circlePoints);
        currentShape = "circle";
      }
    }, 4000);

    return () => clearInterval(interval);
  }, []);

  return (
    <div className="flex flex-col items-center justify-center min-h-screen bg-gray-900">
      <h1 className="text-white mb-8 text-2xl font-bold">GSAP Particle Morphing</h1>
      <canvas
        ref={canvasRef}
        className="border border-gray-700 rounded-lg shadow-2xl bg-black"
      />
      <p className="text-gray-400 mt-4">Every 4 seconds, shapes will change...</p>
    </div>
  );
};

export default ParticleMorph;

"use client";
import React, { useEffect, useRef } from "react";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";

gsap.registerPlugin(ScrollTrigger);

const InteractiveMorph = () => {
  const canvasRef = useRef(null);
  const particles = useRef([]);
  const mouse = useRef({ x: -1000, y: -1000 });
  const numParticles = 800;

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext("2d");
    
    // Canvas size-ah window size-ku match panrom
    const resize = () => {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    };
    window.addEventListener("resize", resize);
    resize();

    // 1. Particle Class (Ovvoru dot-kum interaction logic)
    class Particle {
      constructor(x, y) {
        this.x = x;
        this.y = y;
        this.targetX = x;
        this.targetY = y;
        this.vx = 0;
        this.vy = 0;
        this.radius = 2;
        this.friction = 0.9;
        this.spring = 0.05;
      }

      update() {
        // Mouse Interaction (Repel Effect)
        let dx = mouse.current.x - this.x;
        let dy = mouse.current.y - this.y;
        let distance = Math.sqrt(dx * dx + dy * dy);
        let force = (100 - distance) / 100; // 100px range-kulla repel aagum

        if (distance < 100) {
          this.vx -= dx * force * 0.5;
          this.vy -= dy * force * 0.5;
        }

        // Spring movement (Thirumba target position-ku vara)
        this.vx += (this.targetX - this.x) * this.spring;
        this.vy += (this.targetY - this.y) * this.spring;

        this.vx *= this.friction;
        this.vy *= this.friction;

        this.x += this.vx;
        this.y += this.vy;
      }

      draw() {
        ctx.fillStyle = "#3b82f6";
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fill();
      }
    }

    // 2. Initial Setup
    const centerX = canvas.width / 2;
    const centerY = canvas.height / 2;

    for (let i = 0; i < numParticles; i++) {
      particles.current.push(new Particle(centerX, centerY));
    }

    // 3. Shapes Logic
    const getCirclePoints = () => {
      return Array.from({ length: numParticles }, () => {
        const angle = Math.random() * Math.PI * 2;
        const r = 150 + Math.random() * 20;
        return { x: centerX + Math.cos(angle) * r, y: centerY + Math.sin(angle) * r };
      });
    };

    const getGridPoints = () => {
      return Array.from({ length: numParticles }, () => ({
        x: centerX - 200 + Math.random() * 400,
        y: centerY - 200 + Math.random() * 400,
      }));
    };

    const circlePoints = getCirclePoints();
    const gridPoints = getGridPoints();

    // 4. Mouse Move Event
    const onMouseMove = (e) => {
      mouse.current.x = e.clientX;
      mouse.current.y = e.clientY;
    };
    window.addEventListener("mousemove", onMouseMove);

    // 5. ScrollTrigger - Scroll pannum pothu morph aagum
    const tl = gsap.timeline({
      scrollTrigger: {
        trigger: ".scroll-container",
        start: "top top",
        end: "bottom bottom",
        scrub: 1, // Smooth scrolling transition
      }
    });

    // Morphing transition based on scroll
    particles.current.forEach((p, i) => {
      tl.to(p, { targetX: gridPoints[i].x, targetY: gridPoints[i].y, duration: 1 }, 0);
    });

    // 6. Animation Loop
    const animate = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      particles.current.forEach((p) => {
        p.update();
        p.draw();
      });
      requestAnimationFrame(animate);
    };
    animate();

    return () => {
      window.removeEventListener("resize", resize);
      window.removeEventListener("mousemove", onMouseMove);
      ScrollTrigger.getAll().forEach(t => t.kill());
    };
  }, []);

  return (
    <div className="scroll-container bg-black h-[300vh]"> {/* 300vh na thaan scroll panna space irukkum */}
      <canvas ref={canvasRef} className="fixed top-0 left-0 pointer-events-none" />
      
      <section className="h-screen flex items-center justify-center">
        <h1 className="text-white text-4xl font-bold z-10">Shape 1: Circle (Scroll Down)</h1>
      </section>

      <section className="h-screen flex items-center justify-center">
        <h1 className="text-white text-4xl font-bold z-10">Morphing...</h1>
      </section>

      <section className="h-screen flex items-center justify-center">
        <h1 className="text-white text-4xl font-bold z-10">Shape 2: Grid Square</h1>
      </section>
    </div>
  );
};

export default InteractiveMorph;
